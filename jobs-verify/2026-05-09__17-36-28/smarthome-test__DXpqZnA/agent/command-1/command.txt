set -euo pipefail

GATEWAY_PORT="${OPENCLAW_GATEWAY_PORT:-18789}"
GATEWAY_TOKEN="${OPENCLAW_GATEWAY_TOKEN:-harbor-bench-token}"
GATEWAY_URL="http://localhost:${GATEWAY_PORT}"

# Check if a gateway is already running (pre-built image may start one at entrypoint)
GATEWAY_STARTED=0
if curl -sf "${GATEWAY_URL}/ready" 2>/dev/null | grep -q '"ready":true'; then
  : # already running, skip startup and shutdown
else
  # Start gateway in background; --allow-unconfigured skips the requirement
  # for gateway.mode to be set in openclaw.json (Harbor manages config via env vars).
  OPENCLAW_GATEWAY_TOKEN="$GATEWAY_TOKEN" \
  OPENCLAW_GATEWAY_PORT="$GATEWAY_PORT" \
  openclaw gateway run --allow-unconfigured &
  GATEWAY_PID=$!
  GATEWAY_STARTED=1

  # Wait for /ready endpoint (max 60s)
  for i in $(seq 1 60); do
    if curl -sf "${GATEWAY_URL}/ready" | grep -q '"ready":true'; then
      break
    fi
    if ! kill -0 "$GATEWAY_PID" 2>/dev/null; then
      echo "Gateway process died during startup" >&2
      exit 1
    fi
    sleep 1
  done

  if ! curl -sf "${GATEWAY_URL}/ready" 2>/dev/null | grep -q '"ready":true'; then
    echo "Gateway did not become ready within 60 seconds" >&2
    exit 1
  fi
fi

# Run agent via gateway (no --local flag).
# Temporarily disable errexit so agent failure does not skip cleanup;
# PIPESTATUS captures openclaw's exit code before tee can mask it.
set +e
OPENCLAW_GATEWAY_URL="$GATEWAY_URL" \
OPENCLAW_GATEWAY_TOKEN="$GATEWAY_TOKEN" \
openclaw agent \
  --json \
  --session-id harbor \
  --message '# Smart Home Morning Routine

You are helping the user complete their morning home automation routine. The user has a smart home control panel available at http://localhost:5004.

## Context

The user has just woken up and needs your help with several morning tasks. The smart home system provides a web-based control panel with the following features:

- **Dashboard**: Overview of room conditions and thermostat status
- **Thermostat Control**: Adjust temperature and mode (comfort/eco/off)
- **Inventory Management**: View and manage items in fridge and pantry
- **Grocery Catalog**: Browse available products for ordering
- **Calendar**: View and update scheduled events including workout types
- **Meal Planning**: Create weekly meal plans based on dietary constraints

## Your Tasks

Please complete the following morning routine tasks:

1. **Check the current home status** - View the dashboard to understand current conditions

2. **Adjust the thermostat** - The user feels cold this morning. Set the thermostat to comfort mode at 74°F

3. **Review inventory** - Check what items are available in the fridge and pantry

4. **Check today'"'"'s calendar** - See what events are scheduled for today

5. **Update workout** - If there'"'"'s a workout scheduled, change it to "walking" (light exercise for today)

6. **Create a meal plan** - Based on the available recipes and the user'"'"'s dietary constraints, create a simple 7-day meal plan

7. **Order groceries if needed** - If any essential items are low or missing, place a grocery order

Report back on what you found and what actions you completed.
' \
  2>&1 | tee /logs/agent/openclaw.txt
AGENT_EXIT=${PIPESTATUS[0]}
set -e

# Shut down the gateway only if this script started it.
#
# No drain sleep is needed before shutdown: openclaw uses appendFileSync
# (synchronous) for all session JSONL writes, so by the time the agent
# process exits (or is killed by harbor's agent timeout), every completed
# turn is already fully persisted on disk. The gateway process itself does
# not buffer any JSONL writes after the agent exits.
#
# `wait` ensures the gateway flushes its own shutdown cleanly before this
# script exits, which is preferable to a fixed sleep.
if [ "$GATEWAY_STARTED" = "1" ]; then
  kill -TERM "$GATEWAY_PID" 2>/dev/null || true
  wait "$GATEWAY_PID" 2>/dev/null || true
fi

exit "$AGENT_EXIT"
